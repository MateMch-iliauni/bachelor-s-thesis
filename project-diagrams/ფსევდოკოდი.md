// — გლობალური ცვლადები —
STATES = { IDLE, MONITORING, FIRE_ACTIVE, ALARM }
state ← IDLE
CONST FIRE_ON	<- 80°C
CONST FIRE_OFF	<- 55°C
CONST COLD_FRAMES_REQUIRED    	<- 8       // ~4 წმ @ 2 Hz
CONST COOLING_FRAMES_REQUIRED 	<- 10     // ~5 წმ @ 2 Hz
CONST COOLING_DELTA	<- -0.2°C
CONST BEEP_INTERVAL	<- 10 წმ

// — ინიციალიზაცია —
PROCEDURE Setup():
    init Serial, Buzzer(OFF), Button(INPUT_PULLUP)
    create tempQueue, alertQueue
    recover I2C bus
    IF NOT sensor.begin() THEN halt forever  // შეცდომა
    configure sensor (CHESS, 18-bit, 2 Hz)
    Setup_WiFi() // WiFiManager პორტალი -> დაკავშირება ან offline
    start task SensorTask   on core 1
    start task TelegramTask on core 0

// — მთავარი ნაკადი —
PROCEDURE Loop():
    Update_Buzzer() // ავტომატურად გამორთვა 500 მილიწამის შემდეგ
    IF Button_Pressed() THEN // debounce (20 მილიწამი)
        Stop_Buzzer()
        state ← (state == IDLE) ? MONITORING : IDLE
        reset counters and queues
        RETURN
    IF state == IDLE THEN RETURN
    IF NOT tempQueue.receive(maxTemp) THEN RETURN   // ახალი frame არ არის

    SWITCH state:
        CASE MONITORING:
            IF maxTemp ≥ FIRE_ON THEN
                state ← FIRE_ACTIVE
                peakTemp ← maxTemp ;  previousTemp ← maxTemp
                reset coldFrameCount, coolingFrames

        CASE FIRE_ACTIVE:
            peakTemp ← max(peakTemp, maxTemp)
            delta ← maxTemp − previousTemp
            IF delta ≤ COOLING_DELTA THEN coolingFrames++ ELSE coolingFrames ← 0
            previousTemp ← maxTemp

            // პირობა 1: მკვეთრი ვარდნა ზღვარს ქვემოთ
            IF maxTemp < FIRE_OFF THEN
                coldFrameCount++
                IF coldFrameCount ≥ COLD_FRAMES_REQUIRED THEN Raise_Alarm(maxTemp)
            ELSE
                coldFrameCount ← 0
            // პირობა 2: თანდათანობითი გაგრილების ტენდენცია (ცხელი ჭურჭელი)
            IF coolingFrames ≥ COOLING_FRAMES_REQUIRED THEN Raise_Alarm(maxTemp)

        CASE ALARM:
            IF (now − lastAlertTime) ≥ BEEP_INTERVAL THEN
                Start_Buzzer() ;  Trigger_Alert(maxTemp)   // განმეორებითი შეხსენება

PROCEDURE Raise_Alarm(t):
    state ← ALARM ;  lastAlertTime ← now
    Start_Buzzer()
    Trigger_Alert(t) // t ჩაიწერება alertQueue-ში

// — სენსორის ნაკადი (ბირთვი 1) —
TASK SensorTask():
    LOOP forever:
        IF state == IDLE THEN wait 50 ms ; CONTINUE
        IF sensor.getFrame(frame) fails THEN wait 20 ms ; CONTINUE
        maxT ← max(frame[0 … 767])           // 32×24 პიქსელის უმაღლესი მნიშვნელობა
        tempQueue.overwrite(maxT)           // ყოველთვის უახლესი კადრი
        wait 10 ms

// ── Telegram-ის ნაკადი (ბირთვი 0) ──
TASK TelegramTask():
    LOOP forever:
        IF alertQueue.receive(alert) THEN
            IF WiFi connected THEN
                bot.sendMessage(chatID, alarm_text(alert.temp))   // HTTPS/TLS
        wait 50 ms
